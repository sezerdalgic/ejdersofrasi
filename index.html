class PreloadScene extends Phaser.Scene {

    constructor() { super({ key: 'PreloadScene' }); }

    preload() {

        this.load.on('loaderror', (file) => console.log('Dosya yok:', file.key));

        ['rogue', 'barbarian', 'druid', 'paladin', 'pirate'].forEach(k => {

            this.load.spritesheet(k, `${k}.png`, { frameWidth: 64, frameHeight: 64 });

        });


        this.load.spritesheet('coin', 'coin.png', { frameWidth: 32, frameHeight: 32 });

        this.load.image('potion', 'potion.png');

        this.load.image('logo', 'logo.png');

    }

    create() {

        this.createGenTextures();


        if(this.textures.exists('coin')) {

            this.anims.create({

                key: 'coin_spin',

                frames: this.anims.generateFrameNumbers('coin', { start: 0, end: 15 }),

                frameRate: 16,

                repeat: -1

            });

        }


        ['rogue', 'barbarian', 'druid', 'paladin'].forEach(k => {

            if (this.textures.exists(k)) {

                this.anims.create({

                    key: k + '_walk',

                    frames: this.anims.generateFrameNumbers(k, { start: 0, end: 3 }),

                    frameRate: 8,

                    repeat: -1

                });

            }

        });


        if (this.textures.exists('pirate')) {

            this.anims.create({

                key: 'pirate_walk',

                frames: this.anims.generateFrameNumbers('pirate', { start: 0, end: 3 }),

                frameRate: 8,

                repeat: -1

            });

        }


        this.scene.start('MainScene');

    }

    createGenTextures() {

        const g = this.make.graphics({x:0, y:0});

        g.fillStyle(0x333333);

        g.fillRect(0,0,256,256);

        g.generateTexture('ground_gen', 256, 256);


        const r = this.make.graphics({x:0, y:0});

        r.fillStyle(0xffffff);

        r.fillRect(0,0,32,32);

        r.generateTexture('placeholder_rect', 32, 32);

    }

}


class MainScene extends Phaser.Scene {

    constructor() { super({ key: 'MainScene' }); }


    create() {

        if (!selectedCharKey) return;


        this.ground = this.add.tileSprite(

            0, 0,

            this.scale.width,

            this.scale.height,

            'ground_gen'

        ).setScrollFactor(0).setTint(0x555555);


        this.physics.world.setBounds(-10000, -10000, 20000, 20000);


        const charData = CHARACTERS[selectedCharKey];

        let pKey = this.textures.exists(selectedCharKey) ? selectedCharKey : 'placeholder_rect';

        

        this.player = this.physics.add.sprite(0, 0, pKey);

        if (pKey === 'placeholder_rect') {

            this.player.setTint(charData.color).setScale(2);

        } else {

            this.player.play(selectedCharKey + '_walk');

        }


        this.cameras.main.startFollow(this.player, true, 0.1, 0.1).setZoom(1.2);


        this.hp = charData.hp;

        this.maxHp = charData.hp;

        this.baseDamage = charData.dmg;

        this.ultCharge = 0;

        this.isDead = false;

        score = 0;

        gameTime = 0;


        this.playerLevel = 1;

        this.playerXp = 0;

        this.killCount = 0;


        this.cursors = this.input.keyboard.createCursorKeys();

        this.wasd = this.input.keyboard.addKeys({

            W: 'W', A: 'A', S: 'S', D: 'D',

            SPACE: 'SPACE', SHIFT: 'SHIFT'

        });


        this.enemies = this.physics.add.group();

        this.projectiles = this.physics.add.group();

        this.items = this.physics.add.group();


        this.physics.add.overlap(this.projectiles, this.enemies, this.onHitEnemy, null, this);

        this.physics.add.collider(this.player, this.enemies, this.onHitPlayer, null, this);

        this.physics.add.overlap(this.player, this.items, this.onCollectItem, null, this);


        this.time.addEvent({

            delay: 1000,

            loop: true,

            callback: () => {

                if (!this.isDead) {

                    gameTime++;

                    this.updateHUD();

                }

            }

        });

        this.time.addEvent({

            delay: 800,

            loop: true,

            callback: this.spawnEnemy,

            callbackScope: this

        });

        this.time.addEvent({

            delay: 2000,

            loop: true,

            callback: this.spawnItem,

            callbackScope: this

        });


        this.updateHUD();

    }


    update() {

        if (this.isDead || !this.player) return;


        this.ground.tilePositionX = this.cameras.main.scrollX;

        this.ground.tilePositionY = this.cameras.main.scrollY;


        const speed = CHARACTERS[selectedCharKey].speed;

        let vx = 0, vy = 0;

        if (this.cursors.left.isDown || this.wasd.A.isDown) vx = -speed;

        else if (this.cursors.right.isDown || this.wasd.D.isDown) vx = speed;

        if (this.cursors.up.isDown || this.wasd.W.isDown) vy = -speed;

        else if (this.cursors.down.isDown || this.wasd.S.isDown) vy = speed;

        this.player.setVelocity(vx, vy);

        if (vx !== 0) this.player.setFlipX(vx < 0);


        this.player.setDepth(this.player.y);

        this.enemies.getChildren().forEach(e => e.setDepth(e.y));


        if (Phaser.Input.Keyboard.JustDown(this.wasd.SPACE)) {

            this.basicAttack();

        }

        if (Phaser.Input.Keyboard.JustDown(this.wasd.SHIFT) &&

            this.ultCharge >= CHARACTERS[selectedCharKey].ultReq) {

            this.useUltimate();

        }


        this.enemies.getChildren().forEach(enemy => {

            this.physics.moveToObject(enemy, this.player, 60 + (gameTime / 2));

        });

    }


    useUltimate() {

        this.ultCharge = 0;

        this.updateHUD();


        const char = selectedCharKey;


        if (char === 'paladin') {

            this.player.setVelocity(0, 0);

            if (this.player.body) this.player.body.moves = false;

            this.time.delayedCall(700, () => {

            });

                0.4

                targets: blast,

                alpha: 0,

                duration: 500,

                onComplete: () => blast.destroy()

            });


            const ultDmg = this.baseDamage * 3;

            this.enemies.getChildren().forEach(e => {

                if (Phaser.Math.Distance.Between(

                    this.player.x, this.player.y, e.x, e.y

                ) < radius) {

                    e.hp -= ultDmg;

                    if (e.hp <= 0) this.onEnemyKilled(e);

                }

            });


            const heal = Math.round(this.maxHp * 0.15);

            this.hp = Math.min(this.hp + heal, this.maxHp);

            this.updateHUD();

                    e.hp -= ultDmg;

                        this.player.y,

                        radius,

                        CHARACTERS[char].color,

                        0.25

                    );

                        targets: ring,

                        alpha: 0,

                        duration: 200,

                    });

                            this.player.x, this.player.y, e.x, e.y

                        }

                    });

        } else if (char === 'druid') {

            const radius = 180;

            );

            this.tweens.add({

                targets: ring,

                alpha: 0,

                duration: 500,

                onComplete: () => ring.destroy()

            });


            const ultDmg = this.baseDamage * 2;

            this.enemies.getChildren().forEach(e => {

                if (Phaser.Math.Distance.Between(

                    this.player.x, this.player.y, e.x, e.y

                ) < radius) {

                    e.hp -= ultDmg;

                    if (e.hp <= 0) this.onEnemyKilled(e);

                }

            });


            this.updateHUD();

        }

    }


    spawnItem() {

        const x = this.player.x + Phaser.Math.Between(-400, 400);

        const y = this.player.y + Phaser.Math.Between(-300, 300);


        if (Math.random() > 0.7) {

            let item = this.items.create(x, y, 'potion').setScale(0.8);

            item.type = 'potion';

            this.tweens.add({

                targets: item,

                y: y - 5,

                duration: 1000,

                yoyo: true,

                repeat: -1

            });

        } else {

            let item = this.items.create(

                x, y,

                this.textures.exists('coin') ? 'coin' : 'placeholder_rect'

            );

            if (this.textures.exists('coin')) {

                item.play('coin_spin');

                item.setScale(1.5);

            } else {

                item.setTint(0xffd700);

                item.setDisplaySize(20, 20);

            }

            item.type = 'coin';

        }

    }


    spawnEnemy() {

        if (this.isDead) return;


        const angle = Math.random() * Math.PI * 2;

        const r = 600;

        const x = this.player.x + Math.cos(angle) * r;

        const y = this.player.y + Math.sin(angle) * r;


        let typeKey;

        const roll = Math.random();

        if (roll < 0.4) typeKey = 'light';

        else if (roll < 0.7) typeKey = 'heavy';

        else typeKey = 'gunner';


        const cfg = ENEMY_TYPES[typeKey];


        let enemy = this.enemies.create(

            x,

            y,

            this.textures.exists('pirate') ? 'pirate' : 'placeholder_rect'

        );


        const level = this.playerLevel || 1;

        const hpMult = Math.pow(1.14, level - 1);

        const dmgMult = Math.pow(1.16, level - 1);


        enemy.type = typeKey;

        enemy.maxHp = Math.round(cfg.maxHp * hpMult);

        enemy.hp = enemy.maxHp;

        enemy.damage = Math.round(cfg.damage * dmgMult);

        enemy.xp = cfg.xp;


        enemy.setTint(cfg.tint);


        if (this.textures.exists('pirate')) {

            enemy.play('pirate_walk');

        }

    }


    basicAttack() {

        const offset = this.player.flipX ? -40 : 40;

        const slash = this.add.text(

            this.player.x + offset,

            this.player.y,

            '/',

            { fontSize: '40px', color: '#fff' }

        ).setOrigin(0.5);


        this.tweens.add({

            targets: slash,

            alpha: 0,

            duration: 200,

            onComplete: () => slash.destroy()

        });


        const hit = this.projectiles.create(

            this.player.x + offset,

            this.player.y,

            null

        );

        hit.setVisible(false);

        hit.body.setSize(60, 60);

        hit.damage = this.baseDamage;

        hit.isUlt = false;


        this.time.delayedCall(100, () => hit.destroy());

    }


    onHitEnemy(proj, enemy) {

        enemy.hp -= proj.damage;

        if (!proj.isUlt) proj.destroy();


        if (enemy.hp <= 0) {

            this.onEnemyKilled(enemy);

        }

    }


    onEnemyKilled(enemy) {

        enemy.destroy();


        this.killCount++;


        const gainXp = enemy.xp || 5;

        this.playerXp += gainXp;


        this.ultCharge++;


        this.checkLevelUp();

        this.updateHUD();

    }


    onHitPlayer(pl, en) {

        if (this.isDead) return;


        const dmg = en.damage || 5;

        this.hp -= dmg;


        this.cameras.main.shake(100, 0.005);

        this.updateHUD();


        const a = Phaser.Math.Angle.Between(pl.x, pl.y, en.x, en.y);

        en.x += Math.cos(a) * 50;

        en.y += Math.sin(a) * 50;


        if (this.hp <= 0) this.gameOver();

    }


    onCollectItem(pl, item) {

        if (item.type === 'coin') {

            score += 50;

        } else if (item.type === 'potion') {

            const heal = Math.round(this.maxHp * 0.15);

            this.hp = Math.min(this.hp + heal, this.maxHp);

        }

        item.destroy();

        this.updateHUD();

    }


    checkLevelUp() {

        let nextLevel = this.playerLevel + 1;

        if (nextLevel >= LEVEL_THRESHOLDS.length) return;


        const neededXp = LEVEL_THRESHOLDS[nextLevel];

        if (this.playerXp >= neededXp) {

            this.playerLevel = nextLevel;


            const growth = HERO_GROWTH[selectedCharKey];

            if (growth) {

                this.maxHp = Math.round(this.maxHp * (1 + growth.hp));

                this.baseDamage = Math.round(this.baseDamage * (1 + growth.dmg));

                this.hp = this.maxHp;

            }

        }

    }


    updateHUD() {

        const finalScore = Math.floor(gameTime * 1 + this.killCount * 1.75);

        score = finalScore;


        const scoreEl = document.getElementById('score-display');

        const timeEl  = document.getElementById('time-display');

        const hpFill  = document.getElementById('hp-fill');

        const ultFill = document.getElementById('ult-fill');

        const ultText = document.getElementById('ult-text');

        const xpFill  = document.getElementById('xp-fill');

        const xpText  = document.getElementById('xp-text');

        const lvlInfo = document.getElementById('level-info');


        if (!scoreEl) return;


        scoreEl.innerText = `SKOR: ${score}`;

        timeEl.innerText  = `SÜRE: ${gameTime}s`;


        const hpRatio = this.hp / this.maxHp;

        const hpPct = hpRatio * 100;

        hpFill.style.width = `${Math.max(0, hpPct)}%`;


        const isLowHp = hpRatio <= 0.25;

        hpFill.classList.toggle('hp-critical', isLowHp);

        document.body.classList.toggle('low-hp-active', isLowHp);


        const reqKills = CHARACTERS[selectedCharKey].ultReq;

        const ultPct = Math.min(100, (this.ultCharge / reqKills) * 100);

        ultFill.style.width = `${ultPct}%`;

        ultText.innerText = `ULTI (SHIFT) – ${this.ultCharge} / ${reqKills} KILL`;


        const ultReady = this.ultCharge >= reqKills;

        ultFill.classList.toggle('ulti-ready-glow', ultReady);

        ultText.classList.toggle('ulti-ready-glow', ultReady);


        const currentLevel = this.playerLevel;

        const currentXp = this.playerXp;


        const currentThreshold = LEVEL_THRESHOLDS[currentLevel]     || 0;

        const nextThreshold    = LEVEL_THRESHOLDS[currentLevel + 1] || currentThreshold;


        let xpPct = 100;

        if (nextThreshold > currentThreshold) {

            xpPct = ((currentXp - currentThreshold) / (nextThreshold - currentThreshold)) * 100;

        }

        xpFill.style.width = `${Phaser.Math.Clamp(xpPct, 0, 100)}%`;

        xpText.innerText = `XP: ${currentXp} / ${nextThreshold}`;


        lvlInfo.innerText = `Seviye: ${currentLevel} • Kill: ${this.killCount}`;

    }


    gameOver() {

        this.isDead = true;

        this.physics.pause();

        this.player.setTint(0x555555);


        document.getElementById('hud').style.display = 'none';

        document.getElementById('game-over-screen').style.display = 'block';

        document.getElementById('final-score-text').innerText = `Toplam Puan: ${score}`;

    }

}


function initCharSelection() {

    const container = document.getElementById('char-container');

    Object.keys(CHARACTERS).forEach(key => {

        const char = CHARACTERS[key];

        const div = document.createElement('div');

        div.className = 'char-card';

        div.innerHTML = `

            <div style="width:100%; height:80px; background-color:#${char.color.toString(16)}; margin-bottom:5px;"></div>

            <span style="color:white; font-weight:bold;">${char.name}</span><br>

            <span style="color:#aaa; font-size:11px;">${char.desc.replace('\n','<br>')}</span>

        `;

        div.onclick = () => {

            selectedCharKey = key;

            document.getElementById('char-select-screen').style.display = 'none';

            document.getElementById('hud').style.display = 'flex';

            game = new Phaser.Game({

                type: Phaser.AUTO,

                parent: document.body,

                width: window.innerWidth,

                height: window.innerHeight,

                physics: { default: 'arcade' },

                scene: [PreloadScene, MainScene],

                transparent: true

            });

        };

        container.appendChild(div);

    });

}
